cmake_minimum_required(VERSION 3.20)
project(WildlightCore VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(WILDLIGHT_SANITIZE "Enable address and undefined behavior sanitizers with GCC/Clang" OFF)

add_library(wildlight_simulation src/Simulation.cpp)
target_include_directories(wildlight_simulation PUBLIC include)
if(MSVC)
    target_compile_options(wildlight_simulation PUBLIC /W4 /WX /GR- /EHs-c-)
    target_compile_definitions(wildlight_simulation PUBLIC _HAS_EXCEPTIONS=0)
else()
    target_compile_options(wildlight_simulation PUBLIC -Wall -Wextra -Wpedantic -Werror -fno-exceptions -fno-rtti)
endif()
if(WILDLIGHT_SANITIZE)
    if(MSVC)
        message(FATAL_ERROR "This sanitizer configuration requires GCC/Clang; use a Linux sanitizer run.")
    endif()
    target_compile_options(wildlight_simulation PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(wildlight_simulation PUBLIC -fsanitize=address,undefined)
endif()
add_executable(wildlight_headless tools/headless.cpp)
target_link_libraries(wildlight_headless PRIVATE wildlight_simulation)
include(CTest)
if(BUILD_TESTING)
    add_executable(wildlight_tests tests/simulation_tests.cpp)
    target_link_libraries(wildlight_tests PRIVATE wildlight_simulation)
    add_test(NAME simulation_behaviors COMMAND wildlight_tests)
    add_test(NAME idle_player_match COMMAND wildlight_headless --ticks 600 --seed 101)
    set_tests_properties(simulation_behaviors idle_player_match PROPERTIES TIMEOUT 60)
endif()
